# Post & Comment Mentions Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Independent plan** — no dependency on the upload or chat plans.

**Goal:** Inline `@mention` in post composer and comment input: type `@` to get a suggestion overlay showing people you follow (with global search fallback), tap to insert `@username`, send structured mention IDs alongside the post/comment so the backend can push a notification to each mentioned user.

**Architecture:** A shared `MentionTextField` widget listens to cursor position and text for the `@<query>` pattern, shows an `OverlayEntry` with suggestions sourced from the following list (cached in Riverpod, filtered client-side) with a debounced `GET /users/search` fallback. On submit, the content includes inline `@username` text and a separate `mentionIds: [userId]` array. Backend adds `PostMention` and `CommentMention` join tables, parses mention IDs from the request, and triggers a notification per mentioned user. Content in feeds renders `@username` as a tappable `TextSpan` that navigates to the user's profile. The separate mention text field in `create_post_page.dart` is removed.

**Tech Stack:** Flutter Riverpod, Dart `OverlayEntry`, NestJS, Prisma

```dart
import 'dart:async';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:freebay/core/theme/app_colors.dart';
import 'package:freebay/core/theme/app_typography.dart';
import 'package:freebay/features/social/data/entities/user_search_entity.dart';
import 'package:freebay/features/social/presentation/providers/user_search_provider.dart';

// ── Pure helper functions (exported for testing) ────────────────────────────

String? detectMentionQuery(String text, int cursorPos) {
  final before = text.substring(0, cursorPos);
  final atIndex = before.lastIndexOf('@');
  if (atIndex == -1) return null;
  final between = before.substring(atIndex + 1);
  if (between.contains(' ')) return null;
  return between;
}

({String newText, int newCursor}) replaceMentionInText(
  String text,
  int cursorPos,
  String username,
  String query,
) {
  final before = text.substring(0, cursorPos);
  final atIndex = before.lastIndexOf('@');
  final after = text.substring(cursorPos);
  final newBefore = '${before.substring(0, atIndex)}@$username ';
  return (newText: '$newBefore$after', newCursor: newBefore.length);
}

// ── Widget ───────────────────────────────────────────────────────────────────

class MentionTextField extends ConsumerStatefulWidget {
  final TextEditingController controller;
  final FocusNode? focusNode;
  final String hintText;
  final void Function(List<String> mentionIds) onMentionIdsChanged;
  final int minLines;
  final int? maxLines;

  const MentionTextField({
    super.key,
    required this.controller,
    this.focusNode,
    this.hintText = '',
    required this.onMentionIdsChanged,
    this.minLines = 1,
    this.maxLines,
  });

  @override
  ConsumerState<MentionTextField> createState() => _MentionTextFieldState();
}

class _MentionTextFieldState extends ConsumerState<MentionTextField> {
  OverlayEntry? _overlay;
  final _layerLink = LayerLink();
  final List<String> _mentionIds = [];
  List<UserSearchEntity> _suggestions = [];
  Timer? _debounce;
  String? _activeQuery;

  @override
  void initState() {
    super.initState();
    widget.controller.addListener(_onTextChanged);
  }

  @override
  void dispose() {
    widget.controller.removeListener(_onTextChanged);
    _removeOverlay();
    _debounce?.cancel();
    super.dispose();
  }

  void _onTextChanged() {
    final text = widget.controller.text;
    final cursor = widget.controller.selection.baseOffset;
    if (cursor < 0) return;

    final query = detectMentionQuery(text, cursor);
    if (query == null) {
      _removeOverlay();
      _activeQuery = null;
      return;
    }

    _activeQuery = query;
    _debounce?.cancel();
    _debounce = Timer(const Duration(milliseconds: 300), () => _fetchSuggestions(query));
  }

  Future<void> _fetchSuggestions(String query) async {
    if (!mounted || _activeQuery != query) return;

    // 1. Try following list first (from cache)
    final following = ref.read(followingListProvider);
    final filtered = following
        .where((u) => u.displayName.toLowerCase().contains(query.toLowerCase()))
        .take(5)
        .toList();

    if (filtered.isNotEmpty) {
      _showSuggestions(filtered);
      return;
    }

    // 2. Fallback: global search
    if (query.isEmpty) {
      _removeOverlay();
      return;
    }
    final result = await ref.read(userSearchProvider(query).future);
    if (!mounted || _activeQuery != query) return;
    _showSuggestions(result.take(5).toList());
  }

  void _showSuggestions(List<UserSearchEntity> users) {
    if (users.isEmpty) {
      _removeOverlay();
      return;
    }
    setState(() => _suggestions = users);
    _removeOverlay();
    _overlay = OverlayEntry(builder: (_) => _buildOverlay());
    Overlay.of(context).insert(_overlay!);
  }

  void _removeOverlay() {
    _overlay?.remove();
    _overlay = null;
  }

  void _selectUser(UserSearchEntity user) {
    final text = widget.controller.text;
    final cursor = widget.controller.selection.baseOffset;
    final query = _activeQuery ?? '';
    final (:newText, :newCursor) = replaceMentionInText(text, cursor, user.displayName, query);

    widget.controller.value = TextEditingValue(
      text: newText,
      selection: TextSelection.collapsed(offset: newCursor),
    );

    if (!_mentionIds.contains(user.id)) {
      _mentionIds.add(user.id);
      widget.onMentionIdsChanged(List.unmodifiable(_mentionIds));
    }
    _removeOverlay();
    _activeQuery = null;
  }

  Widget _buildOverlay() {
    return Positioned(
      width: 280,
      child: CompositedTransformFollower(
        link: _layerLink,
        showWhenUnlinked: false,
        offset: const Offset(0, -8),
        targetAnchor: Alignment.topLeft,
        followerAnchor: Alignment.bottomLeft,
        child: Material(
          color: Colors.transparent,
          child: Container(
            color: AppColors.surfaceContainerLowest,
            constraints: const BoxConstraints(maxHeight: 200),
            child: ListView.builder(
              shrinkWrap: true,
              padding: EdgeInsets.zero,
              itemCount: _suggestions.length,
              itemBuilder: (_, i) {
                final user = _suggestions[i];
                return GestureDetector(
                  onTap: () => _selectUser(user),
                  child: Container(
                    padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                    color: Colors.transparent,
                    child: Row(children: [
                      Container(
                        width: 32,
                        height: 32,
                        color: AppColors.surfaceContainerHighest,
                        child: const Icon(Icons.person, size: 18, color: AppColors.mediumGray),
                      ),
                      const SizedBox(width: 10),
                      Text(user.displayName,
                          style: TextStyle(fontFamily: AppTypography.fontFamily, fontSize: 14)),
                    ]),
                  ),
                );
              },
            ),
          ),
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return CompositedTransformTarget(
      link: _layerLink,
      child: TextField(
        controller: widget.controller,
        focusNode: widget.focusNode,
        minLines: widget.minLines,
        maxLines: widget.maxLines ?? (widget.minLines == 1 ? 1 : null),
        decoration: InputDecoration(
          hintText: widget.hintText,
          border: InputBorder.none,
        ),
      ),
    );
  }
}
```

- [ ] **Step 3: Create `frontend/lib/shared/widgets/mention_rich_text.dart`**

```dart
import 'package:flutter/gestures.dart';
import 'package:flutter/material.dart';
import 'package:freebay/core/theme/app_colors.dart';
import 'package:freebay/core/theme/app_typography.dart';

final _mentionRegex = RegExp(r'@\w+');

List<InlineSpan> parseMentionSpans(
  String text,
  void Function(String username) onTap, {
  TextStyle? baseStyle,
}) {
  final spans = <InlineSpan>[];
  int lastEnd = 0;

  for (final match in _mentionRegex.allMatches(text)) {
    if (match.start > lastEnd) {
      spans.add(TextSpan(text: text.substring(lastEnd, match.start), style: baseStyle));
    }
    final username = match.group(0)!;
    spans.add(TextSpan(
      text: username,
      style: (baseStyle ?? const TextStyle()).copyWith(
        color: AppColors.primaryContainer,
        fontWeight: FontWeight.w700,
      ),
      recognizer: TapGestureRecognizer()..onTap = () => onTap(username.substring(1)),
    ));
    lastEnd = match.end;
  }

  if (lastEnd < text.length) {
    spans.add(TextSpan(text: text.substring(lastEnd), style: baseStyle));
  }

  return spans;
}

Widget buildMentionRichText(
  String text,
  BuildContext context, {
  required void Function(String username) onMentionTap,
  TextStyle? style,
  int? maxLines,
  TextOverflow overflow = TextOverflow.ellipsis,
}) {
  return RichText(
    maxLines: maxLines,
    overflow: overflow,
    text: TextSpan(children: parseMentionSpans(text, onMentionTap, baseStyle: style)),
  );
}
```
